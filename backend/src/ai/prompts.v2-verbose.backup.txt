export type Cohort = 'LIGHT' | 'MODERATE' | 'HEAVY';
export type JourneyChapter = 'NHAN_DIEN' | 'KIEM_SOAT' | 'LAM_CHU' | 'TAI_THIET';
export type ChatMode = 'normal' | 'calm' | 'whisper' | 'busy';
export type ChatChannel = 'widget' | 'page';

export interface MentorContext {
  name: string;
  /** Cách Sol gọi user — "anh" | "chị" | "ông" | "bạn" | tuỳ chỉnh. */
  pronouns: string;
  /** Tên hiển thị của bot AI — default "Sol AI". User có thể tuỳ chỉnh
   *  ("Sol Phó tướng", "Sol Vợ yêu", v.v.). KHÔNG nhầm với "Sol Trải nghiệm"
   *  — đó là label cho response từ chip/phím tắt (template, không phải AI). */
  assistantName?: string;

  // ── Hồ sơ cai thuốc ─────────────────────────────────────────────────
  age?: number | null;
  yearsSmoked?: number | null;
  /** 0-5 lý do user tự viết ngày 1. AI replay NGUYÊN VĂN khi user thèm. */
  quitReasons?: string[];

  // ── Cohort & Journey (V2 — canonical 2026-05-18) ────────────────────
  /** LIGHT / MODERATE / HEAVY — null nếu chưa làm FTND. */
  cohort?: Cohort | null;
  /** "NHẸ" | "VỪA" | "NẶNG" (hiển thị). */
  cohortLabel?: string;
  /** Tổng số ngày lộ trình: 35 / 52 / 65. */
  totalDays?: number;
  /** Ngày Q-Day: 15 / 22 / 28 theo cohort. */
  qDay?: number;
  /** Ngày hiện tại trong lộ trình (1..totalDays). */
  dayInJourney?: number;
  /** Trạng thái Q-Day. */
  qDayStatus?: 'before' | 'today' | 'after';
  /** Số ngày đến Q-Day (negative nếu đã qua). */
  daysUntilQDay?: number;

  // ── Chapter ─────────────────────────────────────────────────────────
  chapter?: JourneyChapter;
  chapterLabel?: string;

  // ── LEGACY (giữ tương thích) ────────────────────────────────────────
  /** Deprecated: dùng dayInJourney + totalDays. Giữ cho caller cũ. */
  dayNumber?: number;

  // ── Engagement & FTND ───────────────────────────────────────────────
  ftndScore?: number | null;
  checkinStreak: number;
  topTriggers: string[];
  riskyHours: number[];

  // ── State ───────────────────────────────────────────────────────────
  recentCheckins: Array<{
    dayNumber: number;
    smoked: boolean;
    cravingIntensity: number;
    mood: number;
    note?: string | null;
  }>;
  recentMessages: Array<{ role: 'user' | 'assistant'; content: string }>;
  currentMood: 'improving' | 'declining' | 'stable';

  // ── Channel & Mode ──────────────────────────────────────────────────
  mode: ChatMode;
  /** Widget bubble (chat sidebar) hoặc full page /chat. */
  channel?: ChatChannel;
}
